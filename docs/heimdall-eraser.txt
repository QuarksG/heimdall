// ===================== Hosting path =====================
"Staff user" > "Internet": open Amplify URL
"Admin user" > "Internet": open Amplify URL
"Internet" > "Route 53": DNS resolution
"Route 53" > "CloudFront": route to CDN
"CloudFront" > "AWS WAF": inspect traffic
"AWS WAF" > "CloudFront": allow/block
"CloudFront" > "Amplify Hosting": serve SPA assets
"Amplify Hosting" > "Heimdall SPA": load application

// ===================== Terms acceptance (pre-signup) =====================
"Heimdall SPA" > "API Gateway (Cognito JWT authorizer)": accept terms (unauthenticated route)
"API Gateway (Cognito JWT authorizer)" > "TermsService Lambda (pre-signup)": handle acceptance
"TermsService Lambda (pre-signup)" > "TermsAcceptance (DynamoDB)": PutItem if id not exists

// ===================== Authentication (first gate) =====================
"Heimdall SPA" > "Cognito User Pool": sign up / sign in / forgot password
"Cognito User Pool" > "PreSignUp Lambda (domain allowlist)": validate email domain
"PreSignUp Lambda (domain allowlist)" > "Cognito User Pool": allow or reject signup
"Cognito User Pool" > "Email OTP verification": send OTP
"Email OTP verification" > "Staff user": deliver OTP
"Staff user" > "Cognito User Pool": confirm OTP
"Cognito User Pool" > "Heimdall SPA": issue JWT tokens

// ===================== Role & permission claims =====================
"Cognito User Pool" > "User Pool Groups (Admin/Staff)": group membership
// CHANGED: newest ACTIVE grant only / revoked user gets EMPTY claims / stale-attribute fallback deleted
"Cognito User Pool" > "PreTokenGen Lambda (claims injection)": on login and token refresh
"PreTokenGen Lambda (claims injection)" > "AccessGrants (DynamoDB)": read newest ACTIVE grant
"PreTokenGen Lambda (claims injection)" > "Cognito User Pool": add claims to JWT
"Cognito User Pool" > "Heimdall SPA": updated JWT with claims
"Heimdall SPA" > "usePermissions (RBAC/ABAC)": compute allowed routes/features

// ===================== Access request workflow =====================
"Heimdall SPA" > "API Gateway (Cognito JWT authorizer)": submit access request
"API Gateway (Cognito JWT authorizer)" > "AccessRequestService Lambda": create request
"AccessRequestService Lambda" > "AccessRequests (DynamoDB)": PutItem (PENDING)
"Heimdall SPA" > "API Gateway (Cognito JWT authorizer)": list my requests
"API Gateway (Cognito JWT authorizer)" > "AccessRequestService Lambda": query requests
"AccessRequestService Lambda" > "AccessRequests (DynamoDB)": Query by userId
"Heimdall SPA" > "API Gateway (Cognito JWT authorizer)": admin list pending
"API Gateway (Cognito JWT authorizer)" > "AdminApprovalService Lambda": query pending
"AdminApprovalService Lambda" > "AccessRequests (DynamoDB)": Query by status=PENDING (GSI byStatus NEW)
"Heimdall SPA" > "API Gateway (Cognito JWT authorizer)": approve or reject request
"API Gateway (Cognito JWT authorizer)" > "AdminApprovalService Lambda": process decision
"AdminApprovalService Lambda" > "AccessRequests (DynamoDB)": UpdateItem APPROVED/REJECTED
// CHANGED: soft revocation - one row per grant / mark REVOKED / never delete
"AdminApprovalService Lambda" > "AccessGrants (DynamoDB)": new grant row or mark old REVOKED (never delete)
"Heimdall SPA" > "Cognito User Pool": refresh session
"Cognito User Pool" > "PreTokenGen Lambda (claims injection)": rebuild claims
"PreTokenGen Lambda (claims injection)" > "AccessGrants (DynamoDB)": read updated grants
"Cognito User Pool" > "Heimdall SPA": JWT now includes allowedFeatures

// ===================== Audit trail - activity capture (NEW) =====================
// Browser reports WHAT happened. Server stamps WHO and WHEN from verified JWT claims
"Bounded contexts (src/features)" > "Audit capture (NEW)": file submitted / run completed / result downloaded
"Audit capture (NEW)" > "Hash worker (NEW)": SHA-256 hash file bytes off main thread
"Audit capture (NEW)" > "API Gateway (Cognito JWT authorizer)": POST /audit/events (runId correlation - never blocks user)
"API Gateway (Cognito JWT authorizer)" > "AuditIngestService Lambda (NEW)": verified claims only
"AuditIngestService Lambda (NEW)" > "AuditRecords (DynamoDB append-only) (NEW)": conditional append (sequence + digest)

// ===================== Audit trail - record versioning (NEW) =====================
// DynamoDB Streams fire on COMMITTED writes only - covers every writer including TermsService
"AccessRequests (DynamoDB)" > "VersionRecorder Lambda (NEW)": stream NEW_AND_OLD_IMAGES
"AccessGrants (DynamoDB)" > "VersionRecorder Lambda (NEW)": stream NEW_AND_OLD_IMAGES
"TermsAcceptance (DynamoDB)" > "VersionRecorder Lambda (NEW)": stream NEW_AND_OLD_IMAGES
"VersionRecorder Lambda (NEW)" > "AuditRecords (DynamoDB append-only) (NEW)": append before/after version record

// ===================== Audit trail - read / verify / retention (NEW) =====================
// Browser never touches DynamoDB or Lambda directly - all reads via API Gateway
"Audit console (Admin only) (NEW)" > "API Gateway (Cognito JWT authorizer)": filter / page / export / verify (Admin group)
"API Gateway (Cognito JWT authorizer)" > "AuditQueryService Lambda (NEW)": query audit trail
"AuditQueryService Lambda (NEW)" > "AuditRecords (DynamoDB append-only) (NEW)": Query via GSIs (READ-ONLY)
"API Gateway (Cognito JWT authorizer)" > "AuditVerifier Lambda (NEW)": verify chain integrity for range
"AuditVerifier Lambda (NEW)" > "AuditRecords (DynamoDB append-only) (NEW)": recompute digests and report gaps
"API Gateway (Cognito JWT authorizer)" > "AuditIngestService Lambda (NEW)": record AUDIT_EXPORTED after export
"EventBridge daily schedule (NEW)" > "RetentionManager Lambda (NEW)": trigger daily
"RetentionManager Lambda (NEW)" > "AuditRecords (DynamoDB append-only) (NEW)": purge 730-day expired prefix + purge marker
